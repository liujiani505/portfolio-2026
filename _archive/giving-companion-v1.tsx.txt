/** A product screen in a quiet browser frame. */
function Frame({
  src,
  alt,
  className,
  imgClassName,
  priority,
}: {
  src: string;
  alt: string;
  className?: string;
  imgClassName?: string;
  priority?: boolean;
}) {
  return (
    <div
      className={cn(
 "overflow-hidden rounded-xl border border-ink/10 bg-white shadow-[0_30px_70px_-40px_rgb(40_30_20/0.45)]",
        className,
      )}
    >
      <div className="flex items-center gap-1.5 border-b border-black/10 bg-[#f3f1ee] px-3 py-2" aria-hidden>
        <span className="h-2.5 w-2.5 rounded-full bg-black/15" />
        <span className="h-2.5 w-2.5 rounded-full bg-black/15" />
        <span className="h-2.5 w-2.5 rounded-full bg-black/15" />
      </div>
      {/* eslint-disable-next-line @next/next/no-img-element */}
      <img
        src={src}
        alt={alt}
        loading={priority ? "eager" : "lazy"}
        className={cn("block h-auto w-full", imgClassName)}
      />
    </div>
  );
}

/** A close-up detail cut from a screen. */
function Detail({ src, alt, className }: { src: string; alt: string; className?: string }) {
  return (
    <div className={cn("overflow-hidden rounded-lg border border-ink/10 bg-white shadow-[0_14px_40px_-28px_rgb(40_30_20/0.5)]", className)}>
      {/* eslint-disable-next-line @next/next/no-img-element */}
      <img src={src} alt={alt} loading="lazy" className="block h-auto w-full" />
    </div>
  );
}

function Kicker({ n, children }: { n: string; children: React.ReactNode }) {
  return (
    <p className="eyebrow mb-8" data-n={n}>
      {children}
    </p>
  );
}

function Serif({ children, className, id }: { children: React.ReactNode; className?: string; id?: string }) {
  return (
    <h2 id={id} className={cn("text-balance text-title font-semibold", className)}>
      {children}
    </h2>
  );
}

function Label({ children, tone = "line" }: { children: React.ReactNode; tone?: "line" | "lime" | "ink" }) {
  return (
    <span
      className={cn(
 "inline-flex items-center rounded-full px-3 py-1 text-base font-medium",
        (tone === "lime" || tone === "ink") && "border border-ink/70 text-ink",
        tone === "line" && "border border-line text-muted",
      )}
    >
      {children}
    </span>
  );
}
